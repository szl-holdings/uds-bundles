# This file documents WHERE and HOW the uds-v0.4.0 SDA image is signed and verified.

# SIGNING HAPPENS HERE (founder/Forge step — NOT in this PR):
#   AUTHORITATIVE: keyless via GitHub Actions OIDC (Fulcio + Rekor) in
#                  khipu-sda-core/.github/workflows/cosign.yml — no private key needed.
#   LEGACY/OPT:    keyed ECDSA-P256; PRIVATE key in HF Space secret COSIGN_PRIVATE_KEY
#                  (huggingface.co/spaces/SZLHOLDINGS/<sda-space> → Settings → Secrets).
#                  PUBLIC key: bundles/v0.1.0/cosign_signing_key.pub (this repo).

# --- Verify the container image signature (KEYLESS, authoritative) ---
cosign verify \
  --certificate-identity-regexp "https://github.com/szl-holdings/khipu-sda-core/.*" \
  --certificate-oidc-issuer "https://token.actions.githubusercontent.com" \
  ghcr.io/szl-holdings/khipu-sda-core:uds-v0.4.0

# --- Verify the SLSA provenance attestation (L1 honest; L2 build-attested where CI runs) ---
cosign verify-attestation \
  --certificate-identity-regexp "https://github.com/szl-holdings/khipu-sda-core/.*" \
  --certificate-oidc-issuer "https://token.actions.githubusercontent.com" \
  --type slsaprovenance \
  ghcr.io/szl-holdings/khipu-sda-core:uds-v0.4.0

# --- GitHub-native build provenance (SLSA L2 attestation present) ---
gh attestation verify oci://ghcr.io/szl-holdings/khipu-sda-core:uds-v0.4.0 \
  --owner szl-holdings

# --- Verify any DSSE receipt blob emitted by an SDA anomaly/threat verdict ---
# (Receipts are SAME in-toto Statement v1 schema as szl_mechanics; predicate
#  predicateType = https://szlholdings.com/attestations/sda-anomaly/v1)
cosign verify-blob \
  --certificate-identity-regexp "https://github.com/szl-holdings/khipu-sda-core/.*" \
  --certificate-oidc-issuer "https://token.actions.githubusercontent.com" \
  --signature <receipt>.sig \
  --certificate <receipt>.pem \
  <receipt>.blob
